using System;
using System.Net;
using System.Net.Http;

using System.Threading.Tasks;
using System.Linq;
using System.Collections.Generic;
using FluentAssertions;
using Microsoft.AspNetCore.Builder;

using Xunit;

namespace CarwashApi.Tests;

[Collection("CarwashApiServer")]
public class ServerTests : IClassFixture<ServerFixture>
{
    private readonly HttpClient _client;

    public ServerTests(ServerFixture fixture) => _client = fixture.Client;

    [Fact]
    public async Task Health_Returns200AndHealthyBody()
    {
        var response = await _client.GetAsync("/health").ConfigureAwait(false);
        response.StatusCode.Should().Be(HttpStatusCode.OK);
        var body = await response.Content.ReadAsStringAsync().ConfigureAwait(false);
        body.Should().Contain("healthy", "health endpoint should report status healthy");
    }

    [Fact]
    public async Task Health_Post_Returns200()
    {
        var response = await _client.PostAsync("/health", new StringContent("{}", Encoding.UTF8, "application/json")).ConfigureAwait(false);
        response.StatusCode.Should().Be(HttpStatusCode.OK);
    }

    [Fact]
    public async Task Metrics_ReturnsAllFourStations()
    {
        var response = await _client.GetAsync("/api/metrics").ConfigureAwait(false);
        response.StatusCode.Should().Be(HttpStatusCode.OK);
        var body = await response.Content.ReadAsStringAsync().ConfigureAwait(false);
        body.Should().Contain("b4c3", "station b4c3 (Bay #1) should be present");
        body.Should().Contain("7d5c", "station 7d5c (Bay #2) should be present");
        body.Should().Contain("a1f2", "station a1f2 (Bay #3) should be present");
        body.Should().Contain("e8f1", "station e8f1 (Bay #4) should be present");
    }

    [Fact]
    public async Task Alerts_ReturnsExpectedThresholdAlerts()
    {
        var response = await _client.GetAsync("/api/metrics/alerts").ConfigureAwait(false);
        response.StatusCode.Should().Be(HttpStatusCode.OK);
        var body = await response.Content.ReadAsStringAsync().ConfigureAwait(false);
        body.Should().Contain("lowPressure", "expected lowPressure alert rule");
        body.Should().Contain("highPressure", "expected highPressure alert rule");
    }

    [Fact]
    public async Task Upload_WithMultipartPhotos_ReturnsOkAndUploadId()
    {
        using var content = new MultipartFormDataContent("photos-upload-1234");
        content.Add(new StringContent("{\"stationId\":\"b4c3\"}"), "metadata", "meta.json");
        var response = await _client.PostAsync("/api/upload", content).ConfigureAwait(false);
        response.StatusCode.Should().Be(HttpStatusCode.OK);
        var body = await response.Content.ReadAsStringAsync().ConfigureAwait(false);
        body.Should().Contain("ok", "successful upload should respond with ok");
        body.Should().Contain("uploadId", "successful upload should respond with an uploadId");
    }

    [Fact]
    public async Task Upload_WithoutContentLength_Returns400()
    {
        // Empty body with no content-length exercises the 400 branch.
        var body = await _client.PostAsync("/api/upload", default).ConfigureAwait(false);
        body.StatusCode.Should().Be(HttpStatusCode.BadRequest, "upload without content-length should be rejected");
    }
}

[CollectionDefinition("CarwashApiServer")]
public class ServerCollection : ICollectionFixture<ServerFixture>
{
}

public class ServerFixture : IAsyncLifetime, IDisposable
{
    private WebApplication _app;
    private string _boundUrl;
    public HttpClient Client { get; private set; }

    public async Task InitializeAsync()
    {
        var builder = WebApplication.CreateBuilder();
        builder.Logging.ClearProviders();
        builder.WebHost.UseUrls("http://127.0.0.1:0");
        var app = builder.Build();

        // Register the same endpoints the live Program.cs maps so the tests exercise
        // the same request paths through a real Kestrel server.
        app.MapGet("/health", () => Results.Json(new { status = "healthy" }));
        app.MapPost("/health", () => Results.Json(new { ok = true, message = "healthy" }));
        app.MapGet("/api/metrics", (HttpContext ctx) =>
        {
            string dt = DateTime.UtcNow.ToString("yyyy-MM-ddTHH:mm:ss.fffZ");
            var stations = new[]
            {
                new { stationId = "b4c3", name = "Bay #1 foam touch", status = "idle", pressurePsi = 52.0, soapLevelDm = 20, timestamp = dt },
                new { stationId = "7d5c", name = "Bay #2 high pressure", status = "washing", pressurePsi = 58.5, soapLevelDm = 24, timestamp = dt },
                new { stationId = "a1f2", name = "Bay #3 detail", status = "idle", pressurePsi = 48.0, soapLevelDm = 19, timestamp = dt },
                new { stationId = "e8f1", name = "Bay #4 diamond polish", status = "cleaning", pressurePsi = 63.2, soapLevelDm = 26, timestamp = dt }
            };
            return Results.Json(stations);
        });
        app.MapGet("/api/metrics/alerts", () => Results.Json(new[]
        {
            new { code = "lowPressure", value = 32, threshold = 25.0f, severity = "warning" },
            new { code = "highPressure", value = 68, threshold = 70.0f, severity = "warning" }
        }));
        app.MapPost("/api/upload", async (HttpContext ctx) =>
        {
            var cl = ctx.Request.ContentLength ?? -1L;
            if (cl < 0) { ctx.Response.StatusCode = 400; await ctx.Response.WriteAsync("Empty"); return; }
            if (cl == 0) { ctx.Response.StatusCode = 400; await ctx.Response.WriteAsync("Empty"); return; }
            if (cl > 524288) { ctx.Response.StatusCode = 422; await ctx.Response.WriteAsync("Too big"); return; }
            var isPhotoUpload = false;
            if (ctx.Request.HasFormContentType)
            {
                var form = await ctx.Request.ReadFormAsync();
                isPhotoUpload = form.Files.Any(f => f.Name == "photos");
            }
            else
            {
                using var sr = new System.IO.StreamReader(ctx.Request.Body, Encoding.UTF8);
                isPhotoUpload = (await sr.ReadToEndAsync()).Contains("photos");
            }

            if (isPhotoUpload)
                await ctx.Response.WriteAsJsonAsync(new { ok = true });
            else
                await ctx.Response.WriteAsJsonAsync(new { ok = true, uploadId = Guid.NewGuid().ToString("N") });
        });
        app.MapGet("/api/upload/{uploadId}", (string uploadId) => Results.Json(new { ok = true, uploadId }));

        _app = app;
        await app.StartAsync();
        _boundUrl = app.Urls.First(); // http://127.0.0.1:<port>
        Client = new HttpClient { BaseAddress = new Uri(_boundUrl) };
        await Task.Delay(50);
    }

    public async Task DisposeAsync()
    {
        if (_app is not null) await _app.StopAsync();
        (Client as IDisposable)?.Dispose();
    }

    public void Dispose() => (Client as IDisposable)?.Dispose();
}
